Тестовый класс

// src/test/java/ru/ilyagrg/shop/ShopTest.java
package ru.ilyagrg.shop;

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;

import static org.junit.jupiter.api.Assertions.assertEquals;

class ShopTest {
    private Shop shop;

    @BeforeEach
    void setUp() {
        shop = new Shop();              // поле, а не Shop shop = ...
    }

    @Test
    void totalOfEmptyShopIsZero() {
        assertEquals(0, shop.total());  // ожидаемое, полученное
    }
}

Проверки

assertEquals(ожидаемое, полученное)через equals; числа, строки, Optional, List, Map
assertEquals(Optional.of(90), x)Optional со значением
assertTrue(x.isEmpty())пустой Optional или список
assertFalse(условие)условие ложно
assertSame(a, b)один объект, ==
assertThrows(X.class, () -> код)бросает X; возвращает исключение
IllegalArgumentException e = assertThrows(
        IllegalArgumentException.class,
        () -> shop.addProduct("сыр", -1));
assertEquals("Цена меньше нуля: -1", e.getMessage());

Параметризованные

@ParameterizedTest                        // вместо @Test
@NullSource                               // один запуск с null
@ValueSource(strings = {"", "   "})       // ints = {...} для int
void rejects(String value) { ... }

@ParameterizedTest
@CsvSource({
        "RUB, ₽",                          // одна строка — один запуск
        "USD, $"
})
void knows(Currency c, String symbol) { ... }

Импорты: org.junit.jupiter.params.ParameterizedTest, org.junit.jupiter.params.provider.*.

Четыре вопроса к каждому методу

Обычный случайточное значение, посчитанное заранее
Пусто или нетпустая коллекция, не нашёл
Границаровно на краю условия
Ошибканужное исключение, состояние не изменилось
