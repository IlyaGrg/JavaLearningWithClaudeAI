cart.removeIf(x -> x.equals("Чай"));   // по условию

Map

Map<String, Integer> prices = new HashMap<>();
prices.put("Чай", 250);               // заменяет, возвращает старое или null
prices.get("Чай");                    // нет ключа → null
prices.getOrDefault("Мёд", 0);
prices.containsKey("Чай");   prices.remove("Чай");   prices.size();

for (Map.Entry<String, Integer> e : prices.entrySet()) {
    e.getKey(); e.getValue();
}
prices.keySet();   prices.values();

counts.put(k, counts.getOrDefault(k, 0) + 1);      // подсчёт

Set

Set<String> s = new HashSet<>();
s.add("Илья");        // true — добавлен; false — уже был
s.contains("Илья");   s.remove("Илья");   s.size();
