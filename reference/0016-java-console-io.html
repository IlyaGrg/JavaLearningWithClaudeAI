Цикл команд

try (Scanner scanner = new Scanner(System.in)) {    // один сканер на программу
    while (true) {
        System.out.print("> ");
        if (!scanner.hasNextLine()) {               // Ctrl+D или конец ввода
            break;
        }
        String line = scanner.nextLine();
        if (line.isBlank()) {
            continue;                                // следующий круг
        }
        if (line.trim().equals("exit")) {
            break;
        }
        try {                                        // try ВНУТРИ цикла
            System.out.println(commands.execute(line));
        } catch (BankException | IllegalArgumentException e) {
            System.out.println("Ошибка: " + e.getMessage());
        }
    }
}

Разбор команды

public String execute(String line) {
    String[] parts = line.trim().split("\\s+");      // [команда, арг1, арг2, ...]
    return switch (parts[0]) {
        case "add" -> add(parts);
        case "total" -> "Итого: " + shop.total();
        default -> throw new IllegalArgumentException("Неизвестная команда: " + parts[0]);
    };
}

private String add(String[] parts) {
    requireArgs(parts, 2);                           // 1. число аргументов
    String name = parts[1];                          // 2. достать и преобразовать
    int price = Integer.parseInt(parts[2]);
    shop.addProduct(name, price);                    // 3. логика
    return "Добавлен товар: " + name + ", " + price; // 4. ответ
}

private static void requireArgs(String[] parts, int count) {
    if (parts.length != count + 1) {
        throw new IllegalArgumentException("Команде " + parts[0] + " нужно аргументов: " + count);
    }
}

Ловушки

NoSuchElementException: No line foundnextLine() без hasNextLine() после конца ввода, или System.in уже закрыт другим сканером
split(" ") → [add, , молоко]пиши trim().split("\\s+")
"".split("\\s+").length == 1пустые строки пропускай в Main
ArrayIndexOutOfBoundsExceptionнет requireArgs; не ловится как IllegalArgumentException
For input string: "сто"NumberFormatException — подкласс IllegalArgumentException

catch (RuntimeException e)прячет баги вроде NPE; лови только ошибки пользователя
